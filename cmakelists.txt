cmake_minimum_required(VERSION 3.16)
project(ShopManagement VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6 components
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Sql PrintSupport)

# Enable automatic MOC, UIC and RCC processing
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# Source files
set(SOURCES
    main.cpp
    DatabaseManager.cpp
    Product.cpp
    Supplier.cpp
    Transaction.cpp
    Sale.cpp
    LoginDialog.cpp
    AdminPanel.cpp
    SalesmanPanel.cpp
    SlipPrinter.cpp
)

# Header files
set(HEADERS
    DatabaseManager.h
    Product.h
    Supplier.h
    Transaction.h
    Sale.h
    LoginDialog.h
    AdminPanel.h
    SalesmanPanel.h
    SlipPrinter.h
)

# Create the executable
add_executable(ShopManagement ${SOURCES} ${HEADERS})

# Link Qt6 libraries
target_link_libraries(ShopManagement
    Qt6::Core
    Qt6::Widgets
    Qt6::Sql
    Qt6::PrintSupport
)

# Set target properties
set_target_properties(ShopManagement PROPERTIES
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE TRUE
)

# Installation
install(TARGETS ShopManagement
    BUNDLE DESTINATION .
    RUNTIME DESTINATION bin
)
